sealed class DnaAnalysisOutcome {
    data class Complete(val analysis: DnaAnalysis) : DnaAnalysisOutcome()
    // Deliberately does not expose file paths, URIs or codec exception messages.
    data class Unavailable(val reason: DnaAnalysisFailure, val diagnostic: String? = null) : DnaAnalysisOutcome()
}

/** Shares the waveform decoder's concurrency gate; all work remains offline and off UI. */
@Singleton
class OfflineDnaAnalyzer @Inject constructor(private val pcmSource: PcmSource) {
    suspend fun analyze(context: Context, uri: Uri, durationHintMs: Long = 0L): DnaAnalysisOutcome =
        withContext(Dispatchers.IO) {
            val job = currentCoroutineContext()
            try {
                val analyzer = DnaStreamingAnalyzer(Runnable { job.ensureActive() })
                pcmSource.decode(context, uri, durationHintMs, object : PcmConsumer {
                    override fun onStart(durationUs: Long) = Unit
                    override fun onPcm(buffer: ByteBuffer, offset: Int, size: Int, ptsUs: Long,
                        sampleRate: Int, channels: Int, encoding: Int, checkCancelled: () -> Unit) {
                        checkCancelled()
                        analyzer.accept(buffer, offset, size, ptsUs, sampleRate, channels, encoding)
                    }
                })
                job.ensureActive()
                DnaAnalysisOutcome.Complete(analyzer.finish())
            } catch (cancel: CancellationException) {
                throw cancel
            } catch (error: Exception) {
                DnaAnalysisOutcome.Unavailable(DnaAnalysisFailure.classify(error),
                    (error as? DnaTimestampException)?.diagnostic())
            }
        }
}

import kotlin.math.exp

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — TIME-BASED BAND FOLLOWER + BEAT ENVELOPE
//
// The DSP tap delivers frames at ~22–40Hz; the display refreshes at 60–120Hz.
// The follower runs every RENDER frame and eases the drawn levels toward the
// newest DSP frame with attack/release TIME CONSTANTS (seconds), not per-update
// rates — so the feel is identical at 60Hz and 120Hz and never steps at the
// DSP cadence. Fast attack keeps hits punchy, slower release keeps it silky.
//
// Pure + Android-free (unit-tested on the JVM).
// ─────────────────────────────────────────────────────────────────────────────
class BandFollower(val size: Int) {

    /** Currently drawn levels, 0..1. Read-only by contract. */
    val out = FloatArray(size)

    fun step(target: FloatArray, dtSec: Float, attackSec: Float, releaseSec: Float) {
        val dt = dtSec.coerceIn(0f, MAX_DT)
        val ka = coeff(dt, attackSec)
        val kr = coeff(dt, releaseSec)
        val n = if (target.size < size) target.size else size
        for (i in 0 until n) {
            val t = if (target[i].isFinite()) target[i].coerceIn(0f, 1f) else 0f
            val c = out[i]
            out[i] = c + (t - c) * (if (t > c) ka else kr)
        }
    }

    fun reset() = out.fill(0f)

    companion object {
        const val MAX_DT = 0.1f

        /** 1 - e^(-dt/tau); snaps to the target when tau is ~0. */
        fun coeff(dtSec: Float, tauSec: Float): Float =
            if (tauSec <= 1e-5f) 1f else 1f - exp(-dtSec / tauSec)
    }
}

/**
 * Turns a bass-energy signal into a 0..1 "punch": instant attack on a rise
 * above its own slow average, short release. Drives scale/size/bloom kicks.
 */
class BeatEnvelope {
    private var fast = 0f
    private var slow = 0f

    /** Latest punch value, 0..1. */
    var pulse = 0f
        private set

    fun step(energy: Float, dtSec: Float): Float {
        val dt = dtSec.coerceIn(0f, BandFollower.MAX_DT)
        fast += (energy - fast) * BandFollower.coeff(dt, 0.030f)
        slow += (energy - slow) * BandFollower.coeff(dt, 0.350f)
        val raw = ((fast - slow) * GAIN).coerceIn(0f, 1f)
        pulse = if (raw > pulse) raw else pulse + (raw - pulse) * BandFollower.coeff(dt, 0.120f)
        return pulse
    }

    fun reset() {
        fast = 0f
        slow = 0f
        pulse = 0f
    }

    private companion object {
        const val GAIN = 3.5f
    }
}

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — BAND SOURCES
//
// The renderer asks a BandSource for "the spectrum right now" once per vsync.
// Where the numbers come from (and how latency is compensated) is the source's
// business — so the A/B-able pipelines share one renderer.
//
// CONTRACT: [sample] is called from the render thread only and ALWAYS fills
// [out] (zeros when there is no data). The return value says whether the
// source has usable signal (live, demo, or offline timeline).
// ─────────────────────────────────────────────────────────────────────────────
interface BandSource {
    val bandCount: Int
    fun sample(nowNanos: Long, out: FloatArray): Boolean
}

/** Latency between "analysed in the tap" and "audible from the speaker". */
fun interface LatencyProvider {
    fun latencyNanos(): Long
}

/** Production path: history ring + latency compensation (audio-thread tap). */
class LatencyCompensatedSource(
    private val ring: SpectrumHistoryRing,
    private val latency: LatencyProvider
) : BandSource {
    override val bandCount: Int get() = ring.bandCount
    override fun sample(nowNanos: Long, out: FloatArray): Boolean =
        ring.sample(nowNanos - latency.latencyNanos(), out)
}

/** Legacy / demo path: newest frame from a [SpectrumTripleBuffer], no timing logic. */
class TripleBufferSource(private val bus: SpectrumTripleBuffer) : BandSource {
    private val held = FloatArray(bus.size)
    override val bandCount: Int get() = bus.size
    override fun sample(nowNanos: Long, out: FloatArray): Boolean {
        if (bus.acquire()) System.arraycopy(bus.frontLevels, 0, held, 0, held.size)
        val n = if (out.size < held.size) out.size else held.size
        System.arraycopy(held, 0, out, 0, n)
        return true
    }
}

/** Lets the lab swap pipelines at runtime without touching the renderer. */
class SwitchableSource(initial: BandSource) : BandSource {
    @Volatile var delegate: BandSource = initial
    override val bandCount: Int = initial.bandCount
    override fun sample(nowNanos: Long, out: FloatArray): Boolean = delegate.sample(nowNanos, out)
}

// RIKO DNA (MDNA) — CLASS REACTION CONTRACT (Phase 3)
//
// The per-class REACTION MATRIX: HOW a visual's character changes per sound
// type. A bass section must feel PHYSICALLY different from a vocal section —
// not just louder. Every MDNA visual consumes these profiles:
//
//   LOW_SOUND  — near-still rest state, faint glow, calm
//   DEEP_BASS  — monstrous slow folds + punch, ring flares wide
//   MID_BASS   — groove punch, balanced fold
//   HIGH_BASS  — quicker tight ripples, bright shimmer
//   LOW_VOCAL  — deep slow swell, low pulse
//   VOCAL_ONLY — thin ripple, clean ribs, thin elegant ring, slow breathing
//   FULL_MIX   — everything at once, full corona, maximum motion
//

// character glides when the music section changes — never hard-switches.
// Pure + JVM unit-tested.

    /** Motion speed multiplier (slow swells vs quick ripples). */

    val flare: Float,

    val width: Float,

    /** Slow breathing amplitude. */
    val breathe: Float,

    /** Rib/detail contrast — clean parametric ribs at 1.0. */
    val detail: Float,

    val spin: Float,

    /** Micro-shimmer intensity on the dots. */
    val sparkle: Float

/**
 * Morphs the reaction character toward the active class profile.
 * Renders read [reaction] every frame; the snapshot only allocates while the
 * character is still converging — steady state is allocation-free.
 */
class ClassReactionFollower {
    private var goalClass = SoundClass.LOW_SOUND
    private var current = ClassReaction.forClass(SoundClass.LOW_SOUND)

    /** The morphing reaction state. */
    val reaction: ClassReaction get() = current

    fun step(cls: SoundClass, dtSec: Float) {
        goalClass = cls
        val goal = ClassReaction.forClass(goalClass)
        if (current == goal) return
        val k = 1f - exp(-dtSec.coerceIn(0f, 0.1f) / MORPH_TAU_SEC)
        current = ClassReaction(
            current.fold + (goal.fold - current.fold) * k,
            current.speed + (goal.speed - current.speed) * k,
            current.flare + (goal.flare - current.flare) * k,
            current.width + (goal.width - current.width) * k,
            current.breathe + (goal.breathe - current.breathe) * k,
            current.detail + (goal.detail - current.detail) * k,
            current.spin + (goal.spin - current.spin) * k,
            current.sparkle + (goal.sparkle - current.sparkle) * k
        )
    }

    fun reset() {
        goalClass = SoundClass.LOW_SOUND
        current = ClassReaction.forClass(SoundClass.LOW_SOUND)
    }

    private companion object { const val MORPH_TAU_SEC = 0.22f }
}

/** Render-thread demo clock. It never injects synthetic data into the live pipeline. */
class GenesisDemoSource(override val bandCount: Int = 40) : BandSource {
    @Volatile var paused = false
    private var previous = 0L
    private var seconds = 0f
    @Volatile private var restartClock = false
    fun resumeClock() { restartClock = true }

    override fun sample(nowNanos: Long, out: FloatArray): Boolean {
        if (restartClock) { previous = 0L; restartClock = false }
        if (previous != 0L && !paused) {
            seconds += ((nowNanos - previous) / 1e9f).coerceIn(0f, 0.1f)
        }
        previous = nowNanos
        if (paused) { out.fill(0f); return false }
        SyntheticSpectrum.fill(out, seconds)
        return true
    }
}

/** Bounded frame clock: presentation cadence is distinct from audio timestamps. */
class GenesisFrameClock {
    private var previous = 0L
    var seconds = 0f
        private set
    var intervalMs = 0f
        private set

    fun step(nowNanos: Long, moving: Boolean): Float {
        val dt = if (previous == 0L) 0f else ((nowNanos - previous) / 1e9f).coerceAtLeast(0f)
        previous = nowNanos
        intervalMs = dt * 1000f
        val bounded = dt.coerceAtMost(0.1f)
        if (moving) seconds += bounded
        return bounded
    }
    fun resume() { previous = 0L }
}

package com.rikodev.music.android.dna.core

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — HISTORY CLOCK
//
// Templates like Dot Terrain show the RECENT PAST as rows. Rows are pushed on a
// fixed TIME cadence (not per rendered frame) so the terrain scrolls at the same
// speed at 60Hz, 120Hz or when frames hitch; [frac] (0..1) is how far the next
// row has progressed, letting the shader scroll continuously between pushes.
//
// Pure + Android-free (unit-tested on the JVM).
// ─────────────────────────────────────────────────────────────────────────────
class HistoryClock(val rowIntervalSec: Float = DEFAULT_INTERVAL_SEC) {

    private var acc = 0f

    /** Progress toward the next row, 0 ≤ frac < 1. */
    var frac = 0f
        private set

    /** Advances by [dtSec] and returns how many rows to push now (0..[MAX_ROWS_PER_FRAME]). */
    fun advance(dtSec: Float): Int {
        acc += dtSec.coerceIn(0f, MAX_DT_SEC)
        var rows = 0
        while (acc >= rowIntervalSec && rows < MAX_ROWS_PER_FRAME) {
            acc -= rowIntervalSec
            rows++
        }
        if (acc >= rowIntervalSec) acc = 0f      // never accumulate a backlog
        frac = acc / rowIntervalSec
        return rows
    }

    fun reset() {
        acc = 0f
        frac = 0f
    }

    companion object {
        const val DEFAULT_INTERVAL_SEC = 1f / 24f
        const val MAX_ROWS_PER_FRAME = 4
        private const val MAX_DT_SEC = 0.1f

        /** Float 0..1 → unsigned byte 0..255 (clamped), one history texel per band. */
        fun quantize(levels: FloatArray, out: ByteArray) {
            val n = if (levels.size < out.size) levels.size else out.size
            for (i in 0 until n) {
                val v = levels[i]
                val c = if (v < 0f) 0f else if (v > 1f) 1f else v
                out[i] = (c * 255f + 0.5f).toInt().toByte()
            }
            for (i in n until out.size) out[i] = 0
        }
    }
}

import com.rikodev.music.android.audiofx.SoundClass

/** Render-thread state. No demo clock, decoder, database or offline fallback. */
class LiveDotInput(private val source: BandSource) {
    init { require(source.bandCount == 40) }
    private val target = FloatArray(40)
    private val fastFollower = BandFollower(40)
    private val trailFollower = BandFollower(40)
    val fast: FloatArray get() = fastFollower.out
    val trail: FloatArray get() = trailFollower.out
    var hasSignal = false
        private set

    // ── v2 energy envelopes: asymmetric attack/decay physics ───────────
    // Fast attack keeps hits punchy, slow release glides back to the rim —
    // time constants (not per-frame rates), so 60Hz and 120Hz feel identical.
    private val envTarget = FloatArray(1)
    private val bassFollower = BandFollower(1)
    private val midFollower = BandFollower(1)
    private val punchEnvelope = BeatEnvelope()

    /** Band split for the bass envelope (≈250Hz @ 44.1kHz); renderer updates it. */
    var bassEnd = 11

    /** Band split for the mid envelope (≈4kHz @ 44.1kHz); renderer updates it. */
    var midEnd = 29

    /** Sample rate for band mapping; renderer updates it. */
    var sampleRateHz = 44100

    /** Smoothed bass envelope 0..1 (20Hz–250Hz feel): ~28ms attack, ~200ms release. */
    val bass: Float get() = bassFollower.out[0]

    /** Smoothed mid envelope 0..1 (250Hz–4kHz feel): ~35ms attack, ~220ms release. */
    val mid: Float get() = midFollower.out[0]

    /** Beat punch 0..1 — instant attack on a rise above its own slow average. */
    val punch: Float get() = punchEnvelope.pulse

    /** Accumulating 3D wave phase (radians), wrapped to [0, TAU). */
    var phase = 0f
        private set

    /** Slow 3D roll/twist angle (radians), wrapped to [0, TAU). */
    var roll = 0f
        private set

    // ── Phase 3: sound-class reaction (display domain) ─────────────────
    // Classified from the DISPLAY's own latency-compensated bands, so the
    // visual's character change lines up with what the listener actually
    // hears (the AudioState classifier runs in the audio-time domain and can
    // lead the display by the output latency, e.g. Bluetooth).
    private val classifier = SoundClassifier()
    private val reactions = ClassReactionFollower()
    private var classState = SoundClassState.IDLE

    /** The live sound type of the display-domain spectrum. */
    val soundClass: SoundClass get() = classState.soundClass

    /** The morphing per-class reaction character (fold/flare/breathe/…). */
    val reaction: ClassReaction get() = reactions.reaction

    fun step(nowNanos: Long, dt: Float, playing: Boolean, frozen: Boolean) {
        hasSignal = playing && source.sample(nowNanos, target)
        if (!hasSignal) target.fill(0f)
        // A pause or missing PCM settles to rest even when motion is frozen.
        if (!frozen || !hasSignal) {
            fastFollower.step(target, dt, .018f, .115f)
            trailFollower.step(target, dt, .090f, .240f)
            // Energy envelopes from the followed bands (v^4 power sum, like
            // MdnaSpectrumMath.envelope) — no raw-FFT jitter ever reaches the mesh.
            envTarget[0] = envelope(fastFollower.out, 0, bassEnd)
            bassFollower.step(envTarget, dt, .028f, .200f)
            envTarget[0] = envelope(fastFollower.out, bassEnd, midEnd)
            midFollower.step(envTarget, dt, .035f, .220f)
            punchEnvelope.step(envelope(target, 0, bassEnd), dt)
            // Classify the sound type and morph the visual's character.
            classState = classifier.step(fastFollower.out, sampleRateHz, nowNanos)
            reactions.step(classState.soundClass, dt)
        }
        // The wave only moves while music actually plays — no autonomous drift.
        if (hasSignal && !frozen) {
            phase = wrap(phase + (2.4f + 4.8f * bass) * reaction.speed * dt)
            roll = wrap(roll + (0.10f + 0.30f * mid) * reaction.spin * dt)
        }
    }

    fun reset() {
        target.fill(0f); fastFollower.reset(); trailFollower.reset()
        bassFollower.reset(); midFollower.reset(); punchEnvelope.reset()
        classifier.reset(); reactions.reset(); classState = SoundClassState.IDLE
        phase = 0f; roll = 0f; hasSignal = false
    }

    /** Summed band power (v^4 → ^0.25), perceptual envelope over a band range. */
    private fun envelope(levels: FloatArray, first: Int, end: Int): Float {
        var power = 0.0
        for (i in first.coerceAtLeast(0) until end.coerceAtMost(levels.size)) {
            val v = levels[i]
            if (v.isFinite() && v > 0f) { val d = v.toDouble(); power += d * d * d * d }
        }
        return minOf(1f, power.pow(.25).toFloat())
    }

    private fun wrap(angle: Float): Float {
        val t = angle % TAU
        return if (t < 0f) t + TAU else t
    }

    private companion object { const val TAU = 6.28318530718f }
}

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — QUALITY GOVERNOR
//
// "Smoothest in the world" includes WEAK phones: when frames keep missing the
// display's budget, drop one quality tier instead of stuttering. It only ever
// steps DOWN (no flapping), waits out shader warm-up after start and after every
// change, and ignores one-off hitches (pause/resume, rotation).
//
// Generic over the tier type so it stays pure (unit-tested on the JVM).
// ─────────────────────────────────────────────────────────────────────────────
class QualityGovernor(refreshRateHz: Float) {

    private val budgetMs: Float = 1000f / refreshRateHz.coerceIn(30f, 240f)
    private var sinceChangeMs = 0f
    private var overMs = 0f

    @Volatile var enabled: Boolean = true

    /**
     * Feed one frame time. Returns true exactly when the caller should step
     * DOWN one quality tier now.
     */
    fun onFrame(dtMs: Float, canStepDown: Boolean): Boolean {
        if (dtMs > HITCH_MS) return false          // pause/resume hitch — not a signal
        sinceChangeMs += dtMs
        if (!enabled || !canStepDown) {
            overMs = 0f
            return false
        }
        if (sinceChangeMs < WARMUP_MS) return false
        overMs = if (dtMs > budgetMs * SLOW_FACTOR) overMs + dtMs else (overMs - dtMs * 2f).coerceAtLeast(0f)
        if (overMs >= OVER_LIMIT_MS) {
            overMs = 0f
            sinceChangeMs = 0f
            return true
        }
        return false
    }

    /** Call when quality is changed from outside (user pressed a button). */
    fun reset() {
        sinceChangeMs = 0f
        overMs = 0f
    }

    private companion object {
        const val SLOW_FACTOR = 1.6f
        const val OVER_LIMIT_MS = 1500f
        const val WARMUP_MS = 2500f
        const val HITCH_MS = 250f
    }
}

package com.rikodev.music.android.dna.core

import java.util.concurrent.atomic.AtomicIntegerArray
import java.util.concurrent.atomic.AtomicLong
import java.util.concurrent.atomic.AtomicLongArray

/**
 * Bounded single-producer time-addressed spectrum history. Each slot carries its own
 * sequence stamp: readers reject an in-progress overwrite, even before head advances.
 * Atomic float bits avoid plain-array data races on ARM. No lock or allocation in publish/sample.
 * Each consumer owns its output array; multiple read-only consumers are safe.
 */
class SpectrumHistoryRing(val bandCount: Int, val capacity: Int = 64) {
    init { require(bandCount > 0); require(capacity >= GUARD * 2) }
    private val values = AtomicIntegerArray(capacity * bandCount)
    private val stamps = AtomicLongArray(capacity)
    private val sequences = AtomicLongArray(capacity)
    private val head = AtomicLong(0)
    private val floor = AtomicLong(Long.MIN_VALUE)

    fun publish(src: FloatArray, timestampNanos: Long) {
        val h = head.get()
        val slot = slotOf(h)
        sequences.set(slot, h * 2 + 1) // odd: in progress
        for (i in 0 until bandCount) {
            values.set(slot * bandCount + i, (if (i < src.size) src[i] else 0f).toRawBits())
        }
        stamps.set(slot, timestampNanos)
        sequences.set(slot, h * 2 + 2) // even: complete
        head.set(h + 1)
    }

    /** Monotonic floor prevents a concurrent older reset from resurrecting stale frames. */
    fun markReset(nowNanos: Long) {
        var old = floor.get()
        while (nowNanos > old && !floor.compareAndSet(old, nowNanos)) old = floor.get()
    }

    fun sample(targetNanos: Long, out: FloatArray): Boolean {
        repeat(3) {
            val h = head.get()
            if (h == 0L) return none(out)
            val reset = floor.get()
            val newest = h - 1
            val latestTime = stamps.get(slotOf(newest))
            if (latestTime < reset || targetNanos - latestTime > STALE_NS) return none(out)
            val oldest = (h - (capacity - GUARD)).coerceAtLeast(0L)
            var a = newest
            while (a > oldest && stamps.get(slotOf(a)) > targetNanos) a--
            var b = (a + 1).coerceAtMost(newest)
            val ta = stamps.get(slotOf(a))
            val tb = stamps.get(slotOf(b))
            if (ta < reset) a = b
            if (ta >= targetNanos || a == newest) b = a
            val sa = slotOf(a)
            val sb = slotOf(b)
            val va = a * 2 + 2
            val vb = b * 2 + 2
            if (sequences.get(sa) != va || sequences.get(sb) != vb) return@repeat
            val t = if (a == b) 0f else
                ((targetNanos - ta).toFloat() / (tb - ta).coerceAtLeast(1L)).coerceIn(0f, 1f)
            for (i in out.indices) {
                out[i] = if (i >= bandCount) 0f else {
                    val x = Float.fromBits(values.get(sa * bandCount + i))
                    val y = Float.fromBits(values.get(sb * bandCount + i))
                    x + (y - x) * t
                }
            }
            if (sequences.get(sa) == va && sequences.get(sb) == vb && floor.get() == reset &&
                stamps.get(sa) >= reset && stamps.get(sb) >= reset) return true
        }
        return none(out)
    }

    private fun slotOf(index: Long) = (index % capacity).toInt()
    private fun none(out: FloatArray): Boolean { out.fill(0f); return false }

    companion object {
        const val GUARD = 4
        const val STALE_NS = 300_000_000L
    }
}

package com.rikodev.music.android.dna.core

import java.util.concurrent.atomic.AtomicInteger

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — SPECTRUM TRIPLE BUFFER
//
// Lock-free, allocation-free handoff of the newest spectrum frame from ONE
// producer thread to ONE consumer thread (the GL render thread).
//
//   - No StateFlow, no Compose recomposition, no locks, no per-frame garbage.
//   - The consumer ALWAYS reads a complete frame (no tearing) and never blocks
//     the producer; the producer never blocks the consumer.
//   - Frames the consumer was too slow to read are simply replaced by newer
//     ones (latest-wins), which is exactly what a renderer wants.
//
// CONTRACT: exactly one thread calls [publish], exactly one thread calls
// [acquire] / reads [frontLevels]. Pure + Android-free so the semantics are
// unit-tested on the JVM (SpectrumTripleBufferTest).
// ─────────────────────────────────────────────────────────────────────────────
class SpectrumTripleBuffer(val size: Int) {

    init {
        require(size > 0) { "size must be positive" }
    }

    private val slots = Array(3) { FloatArray(size) }
    private val stamps = LongArray(3)

    // Producer-owned slot (being written).
    private var back = 0

    // Consumer-owned slot (being read).
    private var front = 2

    // Shared slot: low 2 bits = slot index, bit 2 = "fresh, not yet consumed".
    private val middle = AtomicInteger(1)

    /** Producer thread only. Copies [src] (zero-padded / truncated to [size]). */
    fun publish(src: FloatArray, timestampNanos: Long) {
        val dst = slots[back]
        val n = if (src.size < size) src.size else size
        System.arraycopy(src, 0, dst, 0, n)
        if (n < size) java.util.Arrays.fill(dst, n, size, 0f)
        stamps[back] = timestampNanos
        val prev = middle.getAndSet(back or DIRTY)
        back = prev and INDEX_MASK
    }

    /**
     * Consumer thread only. Returns true when a newer frame was swapped in;
     * [frontLevels] / [frontTimestampNanos] then describe it. Returns false
     * (and leaves the front frame untouched) when nothing new was published.
     */
    fun acquire(): Boolean {
        if ((middle.get() and DIRTY) == 0) return false
        val prev = middle.getAndSet(front)
        front = prev and INDEX_MASK
        return true
    }

    /** Consumer thread only. Read-only by contract; valid until the next [acquire]. */
    val frontLevels: FloatArray get() = slots[front]

    /** Consumer thread only. Producer-supplied timestamp of [frontLevels]. */
    val frontTimestampNanos: Long get() = stamps[front]

    private companion object {
        const val DIRTY = 0b100
        const val INDEX_MASK = 0b011
    }
}

package com.rikodev.music.android.dna.core

import kotlin.math.exp
import kotlin.math.sin

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — SYNTHETIC SPECTRUM (lab / demo only)
//
// A deterministic 120BPM "song": kick on the beat, hats on the off-beat and a
// slow melodic peak sweeping across the bands. Lets the visual be judged (and
// profiled) with no music playing. Never used for real playback visuals.
// ─────────────────────────────────────────────────────────────────────────────
object SyntheticSpectrum {

    fun fill(out: FloatArray, timeSec: Float) {
        val n = out.size
        if (n == 0) return
        val beat = timeSec * 2f                      // 120 BPM
        val kickPhase = beat - kotlin.math.floor(beat)
        val kick = exp(-kickPhase * 7f)
        val offPhase = (beat + 0.5f) - kotlin.math.floor(beat + 0.5f)
        val hat = exp(-offPhase * 14f)
        val center = 0.5f + 0.35f * sin(timeSec * 0.4f)
        val vib = 0.6f + 0.4f * sin(timeSec * 3f)
        for (i in 0 until n) {
            val x = if (n > 1) i / (n - 1f) else 0f
            val bassShape = exp(-(x / 0.16f) * (x / 0.16f))
            val dm = (x - center) / 0.08f
            val melody = 0.55f * exp(-dm * dm) * vib
            val hatShape = if (x > 0.6f) (x - 0.6f) / 0.4f else 0f
            val v = 0.06f + 0.85f * kick * bassShape + melody + 0.45f * hat * hatShape
            out[i] = v.coerceIn(0f, 1f)
        }
    }
}

@javax.inject.Singleton
class DnaRepository @Inject constructor(
    @ApplicationContext private val context: Context,
    private val analyzer: OfflineDnaAnalyzer,
    private val profiles: DnaProfiles
) {
    private val gate = Mutex()

    /** One attempt returns the genome and timeline together; failure never triggers a second decode. */
    suspend fun load(song: Song): DnaProfileResult = gate.withLock { withContext(Dispatchers.IO) {
        try {
            val key = song.dnaKey()
            val cached = profiles.load(key)
            if (cached != null) {
                val timeline = try {
                    android.util.AtomicFile(timelineFile(key)).openRead().use { DnaTimelineCodec.read(it) }
                } catch (cancel: CancellationException) { throw cancel }
                catch (_: java.io.IOException) { null }
                catch (_: IllegalArgumentException) { null }
                // A missing/interrupted/corrupt cache can be rebuilt without deleting the profile/history.
                if (timeline != null) return@withContext DnaProfileResult.Ready(cached, timeline)
            }
            when (val result = analyzer.analyze(context, song.uri, song.duration)) {
                is DnaAnalysisOutcome.Unavailable -> DnaProfileResult.Unavailable(result.reason, result.diagnostic)
                is DnaAnalysisOutcome.Complete -> {
                    val a = result.analysis
                    val genome = DnaGenomeMapper.generate(a.audioSha256, a.features, DnaAlgorithmVersion.GENESIS_V1)
                    val atomic = android.util.AtomicFile(timelineFile(key))
                    val stream = atomic.startWrite()
                    try { DnaTimelineCodec.write(a.timeline, stream); atomic.finishWrite(stream) }
                    catch (error: Throwable) { atomic.failWrite(stream); throw error }
                    profiles.storeIfAbsent(key, a.features, genome, song)
                    DnaProfileResult.Ready(requireNotNull(profiles.load(key)), a.timeline)
                }
            }
        } catch (cancel: CancellationException) { throw cancel }
        catch (_: Exception) { DnaProfileResult.Unavailable(DnaAnalysisFailure.STORAGE) }
    } }

    suspend fun genome(song: Song): DnaGenome? = (load(song) as? DnaProfileResult.Ready)?.genome
    suspend fun timeline(song: Song): DnaTimeline? = (load(song) as? DnaProfileResult.Ready)?.timeline

// LIVE·DOT SPECTRUM v3 (Phase 6 — THE CROWN JEWEL, target video match)
//
// The hero visual: a concentric polar dot-matrix mesh living inside a neon
// corona ring. Rest state = all dots merged at the outer rim (thick bright
// border, empty dark center). On beat, 5 vibration zones erupt on the ring,
// sending waves crashing inward that fold the mesh like 3D fabric — dots
// overlap into bright ribbons where the fold is deep, spread into individual
// points where it is shallow. Between beats the mesh snaps back to the rim.
//
// TARGET VIDEO BEHAVIOR (verified from frame-by-frame summary):
//   00:00-00:01 rest:      dots at rim, only bright ring visible
//   00:01-00:03 beat drop: 5 vibration points erupt → wave rushes inward
//   00:02 peak:            3D dome/sphere mesh fills the circle
//   00:05-00:11 folding:   mesh folds/twists like fabric, depth at center
//   00:04/00:08/00:11 retraction: dots snap back to rim between beats
//
// TECHNIQUE:
//   - Concentric polar grid (48 rings × columns), edge-clamped at rest
//   - Inward wave propagation with per-ring phase delay (traveling wave)
//   - 5 beat-vibration zones on the rim (uPunch-driven, localized)
//   - 3D z-displacement + pitch/roll rotation → fabric fold overlap
//   - Additive blending: overlap = bright neon ribbons (bloom effect)
//   - Fast retraction: BeatEnvelope snap + ringFactor edge-clamp
// ─────────────────────────────────────────────────────────────────────────────

/** CPU-side spec of the LIVE·DOT laws — the VERT shader mirrors these formulas. */
object LiveDotMath {
    const val VIBRATION_ZONES = 5

    /** Rest radius — edge-clamped, all rings near outer rim at rest. */
    fun restRadius(ringFactor: Float): Float = 1f - 0.06f * ringFactor

    /** Inward fold depth — bass + vibration push shells inward. */
    fun foldDepth(body: Float, ringFactor: Float, fold: Float, vibration: Float): Float =
        body * ringFactor * (0.45f + 0.55f * vibration) * (0.6f + 0.4f * fold)

    /** Wave amplitude — smooth harmonics scaled by energy and class. */
    fun waveAmplitude(body: Float, mids: Float, ringFactor: Float, fold: Float): Float =
        0.18f * (0.4f + 0.6f * mids) * ringFactor * (0.15f + 0.85f * body) * (0.6f + 0.4f * fold)

    /** 3D fabric fold depth. */
    fun depth(body: Float, ringFactor: Float, fold: Float): Float =
        body * ringFactor * fold

    /**
     * Beat vibration at an angle — localized punch zones on the rim.
     * Strongest at the outer edge (ringFactor=0), fades inward.
     * Creates the "4-6 vibration points" from the target video.
     */
    fun vibration(angle: Float, punch: Float, ringFactor: Float): Float {
        val zoneStep = (2.0 * PI / VIBRATION_ZONES).toFloat()
        val zoneAngle = floor(angle / zoneStep) * zoneStep
        val dist = abs(angle - zoneAngle)
        val rimWeight = (1f - ringFactor) * (1f - ringFactor)
        return punch * exp(-dist * 12f) * rimWeight
    }

    /** Inward propagation phase delay — outer rings react first. */
    fun propagationDelay(v: Float): Float = v * 2.5f
}

    fun amplitude(energy: Float, body: Float, fold: Float): Float =

/**

 * stepping, standard uniform upload and the three-pass draw.
 * Subclasses supply [vert] and [rows] only.
 */
abstract class BaseVisualRenderer(
    private val source: BandSource,
    refreshRateHz: Float = 60f
) : GLSurfaceView.Renderer {
    init { require(source.bandCount == SpectrumBands.COUNT) }

    /** The visual's vertex shader — geometry, light and colour per aId. */
    protected abstract val vert: String

    /** Dot-grid rows; columns = points / rows. Must divide every tier size. */
    protected abstract val rows: Int

    /** Shared glow fragment shader; override only for exotic shapes. */
    protected open val frag: String = VisualDots.FRAG

    // ── Public tuning — same contract as LiveDotRenderer (hosts set these) ──
    @Volatile var quality = MdnaQuality.MID
    @Volatile var autoQuality = true
    @Volatile var playing = true
    @Volatile var frozen = false
    @Volatile var palette = 0f
    @Volatile var seed = 0f
    @Volatile var sensitivity = 1.5f
    @Volatile var sampleRateHz = 44100
    @Volatile var displayFrameNanos = 0L
    @Volatile var error: String? = null
        private set
    @Volatile var frameMs = 0f
        private set
    @Volatile var hasSignal = false
        private set
    @Volatile var bass = 0f
        private set
    @Volatile var mid = 0f
        private set
    @Volatile var treble = 0f
        private set

    protected val input = LiveDotInput(source)
    private val governor = QualityGovernor(refreshRateHz)
    private var visualProgram = 0
    private var bgProgram = 0
    private val buffers = IntArray(2) // [0] backdrop ids, [1] visual ids
    private var visualIdAttribute = 0
    private var bgIdAttribute = 0
    private val uniforms = IntArray(VisualContract.STANDARD_UNIFORMS.size)
    private val bgUniforms = IntArray(5)
    private var width = 1
    private var height = 1
    private var lastFrame = 0L
    private var timeAccum = 0f
    private var previousQuality = quality
    private var maxPointSize = 1f

    /** Points for a quality tier. Divisible by [rows] in every tier. */
    protected open fun pointsFor(quality: MdnaQuality): Int = when (quality) {
        MdnaQuality.LOW -> 3840
        MdnaQuality.MID -> 9216
        MdnaQuality.HIGH -> MAX_POINTS
    }

    override fun onSurfaceCreated(gl: GL10?, config: EGLConfig?) {
        visualProgram = 0; bgProgram = 0; buffers[0] = 0; buffers[1] = 0
        error = null; lastFrame = 0L
        input.reset(); governor.reset(); frameMs = 0f
        timeAccum = 0f
        hasSignal = false; bass = 0f; mid = 0f; treble = 0f
        // Compile backdrop first — keep it even if visual shader fails.
        try {
            bgProgram = link(VisualBackdrop.VERT, VisualBackdrop.FRAG)
            val bgNames = arrayOf("uViewport", "uPointSize", "uMaxPointSize", "uSeed", "uPhase")
            for (i in bgNames.indices) bgUniforms[i] = GLES20.glGetUniformLocation(bgProgram, bgNames[i])
            bgIdAttribute = GLES20.glGetAttribLocation(bgProgram, "aId")
            check(bgIdAttribute >= 0 && bgUniforms.all { it >= 0 }) { "Backdrop shader input missing" }
            buffers[0] = makeIdBuffer(BACKGROUND_POINTS)
        } catch (failure: Exception) {
            error = "Backdrop: ${failure.message}"
        }
        // Compile visual shader — failure keeps backdrop alive.
        try {
            visualProgram = link(vert, frag)
            for (i in VisualContract.STANDARD_UNIFORMS.indices) {
                uniforms[i] = GLES20.glGetUniformLocation(visualProgram, VisualContract.STANDARD_UNIFORMS[i])
            }
            visualIdAttribute = GLES20.glGetAttribLocation(visualProgram, "aId")
            check(visualIdAttribute >= 0) { "Visual shader missing aId" }
            check(uniforms[VisualContract.U_VIEWPORT] >= 0 && uniforms[VisualContract.U_PASS] >= 0) {
                "Visual shader missing core uniforms"
            }
            buffers[1] = makeIdBuffer(MAX_POINTS)
            val range = FloatArray(2)
            GLES20.glGetFloatv(GLES20.GL_ALIASED_POINT_SIZE_RANGE, range, 0)
            maxPointSize = range[1].coerceAtLeast(1f)
            GLES20.glDisable(GLES20.GL_DEPTH_TEST)
            GLES20.glEnable(GLES20.GL_BLEND)
            GLES20.glBlendFunc(GLES20.GL_ONE, GLES20.GL_ONE)
            check(GLES20.glGetError() == GLES20.GL_NO_ERROR) { "Visual GPU initialization failed" }
        } catch (failure: Exception) {
            if (error == null) error = "Visual: ${failure.message}"
        }
    }

    override fun onSurfaceChanged(gl: GL10?, width: Int, height: Int) {
        this.width = width.coerceAtLeast(1); this.height = height.coerceAtLeast(1)
        GLES20.glViewport(0, 0, this.width, this.height)
    }

    override fun onDrawFrame(gl: GL10?) {
        GLES20.glClearColor(0f, 0f, 0f, 1f)
        GLES20.glClear(GLES20.GL_COLOR_BUFFER_BIT)
        if (bgProgram == 0) return

        val now = displayFrameNanos.takeIf { it > 0L } ?: System.nanoTime()
        val elapsed = if (lastFrame == 0L) 0f else (now - lastFrame).coerceAtLeast(0L) / 1e9f
        lastFrame = now
        if (elapsed in 0.001f..0.25f) {
            frameMs = if (frameMs == 0f) elapsed * 1000f else frameMs + (elapsed * 1000f - frameMs) * .08f
        }
        if (elapsed in 0f..0.25f) timeAccum += elapsed
        if (previousQuality != quality) { governor.reset(); previousQuality = quality }
        governor.enabled = autoQuality && playing && !frozen
        if (governor.onFrame(elapsed * 1000f, quality != MdnaQuality.LOW)) quality = quality.lower() ?: quality

        val bassEnd = SpectrumBands.bandIndex(250f, sampleRateHz.coerceAtLeast(8000))
        val midEnd = SpectrumBands.bandIndex(4000f, sampleRateHz.coerceAtLeast(8000))
        input.bassEnd = bassEnd
        input.midEnd = midEnd
        input.sampleRateHz = sampleRateHz
        input.step(now, elapsed, playing, frozen)
        hasSignal = input.hasSignal
        bass = MdnaSpectrumMath.envelope(input.fast, 0, bassEnd)
        mid = MdnaSpectrumMath.envelope(input.fast, bassEnd, midEnd)
        treble = MdnaSpectrumMath.envelope(input.fast, midEnd, 40)

        // Pass 0 — shared starfield + nebula.
        GLES20.glUseProgram(bgProgram)
        GLES20.glUniform2f(bgUniforms[0], width.toFloat(), height.toFloat())
        GLES20.glUniform1f(bgUniforms[1], pointSize())
        GLES20.glUniform1f(bgUniforms[2], maxPointSize)
        GLES20.glUniform1f(bgUniforms[3], seed.coerceIn(0f, 1f))
        GLES20.glUniform1f(bgUniforms[4], input.phase)
        bindIds(buffers[0], bgIdAttribute)
        GLES20.glDrawArrays(GLES20.GL_POINTS, 0, BACKGROUND_POINTS)
        GLES20.glDisableVertexAttribArray(bgIdAttribute)

        // Passes 1+2 — the visual's halo and core (skip if shader failed).
        if (visualProgram != 0 && error == null) {
            val points = pointsFor(quality)
        GLES20.glUseProgram(visualProgram)
        GLES20.glUniform2f(uniforms[VisualContract.U_VIEWPORT], width.toFloat(), height.toFloat())
        GLES20.glUniform1f(uniforms[VisualContract.U_COLUMNS], points / rows.toFloat())
        GLES20.glUniform1f(uniforms[VisualContract.U_POINT_SIZE], pointSize())
        GLES20.glUniform1f(uniforms[VisualContract.U_MAX_POINT_SIZE], maxPointSize)
        GLES20.glUniform1f(uniforms[VisualContract.U_PALETTE], palette.coerceIn(0f, 1f))
        GLES20.glUniform1f(uniforms[VisualContract.U_SEED], seed.coerceIn(0f, 1f))
        GLES20.glUniform1f(uniforms[VisualContract.U_GAIN], sensitivity.coerceIn(.5f, 3f))
        GLES20.glUniform1f(uniforms[VisualContract.U_BASS], input.bass)
        GLES20.glUniform1f(uniforms[VisualContract.U_MID], input.mid)
        GLES20.glUniform1f(uniforms[VisualContract.U_PUNCH], input.punch)
        GLES20.glUniform1f(uniforms[VisualContract.U_PHASE], input.phase)
        GLES20.glUniform1f(uniforms[VisualContract.U_ROLL], input.roll)
        GLES20.glUniform1f(uniforms[VisualContract.U_TIME], timeAccum)
        val reaction = input.reaction
        GLES20.glUniform1f(uniforms[VisualContract.U_FOLD], reaction.fold)
        GLES20.glUniform1f(uniforms[VisualContract.U_FLARE], reaction.flare)
        GLES20.glUniform1f(uniforms[VisualContract.U_WIDTH], reaction.width)
        GLES20.glUniform1f(uniforms[VisualContract.U_BREATHE], reaction.breathe)
        GLES20.glUniform1f(uniforms[VisualContract.U_DETAIL], reaction.detail)
        GLES20.glUniform1f(uniforms[VisualContract.U_SPARKLE], reaction.sparkle)
        GLES20.glUniform1fv(uniforms[VisualContract.U_BANDS], 40, input.fast, 0)
        GLES20.glUniform1fv(uniforms[VisualContract.U_TRAIL], 40, input.trail, 0)
        bindIds(buffers[1], visualIdAttribute)
        GLES20.glUniform1f(uniforms[VisualContract.U_PASS], 1f)
        GLES20.glDrawArrays(GLES20.GL_POINTS, 0, points)
        GLES20.glUniform1f(uniforms[VisualContract.U_PASS], 2f)
        GLES20.glDrawArrays(GLES20.GL_POINTS, 0, points)
        GLES20.glDisableVertexAttribArray(visualIdAttribute)
        } // end visual draw

        if (GLES20.glGetError() != GLES20.GL_NO_ERROR) {
            error = "Visual GPU draw failed"; release()
        }
    }

    /** GL thread, while the owning context is current. */
    fun release() {
        if (visualProgram != 0) GLES20.glDeleteProgram(visualProgram)
        if (bgProgram != 0) GLES20.glDeleteProgram(bgProgram)
        GLES20.glDeleteBuffers(2, buffers, 0)
        visualProgram = 0; bgProgram = 0; buffers[0] = 0; buffers[1] = 0
    }

    // ── internals ────────────────────────────────────────────────────────

    private fun pointSize(): Float = (minOf(width, height) / 260f).coerceIn(1.6f, 6f)

    private fun makeIdBuffer(count: Int): Int {
        val vertices = ByteBuffer.allocateDirect(count * 4).order(ByteOrder.nativeOrder()).asFloatBuffer()
        for (i in 0 until count) vertices.put(i.toFloat())
        vertices.flip()
        val handle = IntArray(1)
        GLES20.glGenBuffers(1, handle, 0)
        GLES20.glBindBuffer(GLES20.GL_ARRAY_BUFFER, handle[0])
        GLES20.glBufferData(GLES20.GL_ARRAY_BUFFER, count * 4, vertices, GLES20.GL_STATIC_DRAW)
        return handle[0]
    }

    private fun bindIds(buffer: Int, attribute: Int) {
        GLES20.glBindBuffer(GLES20.GL_ARRAY_BUFFER, buffer)
        GLES20.glEnableVertexAttribArray(attribute)
        GLES20.glVertexAttribPointer(attribute, 1, GLES20.GL_FLOAT, false, 4, 0)
    }

    private fun link(vertCode: String, fragCode: String): Int {
        val vs = compile(GLES20.GL_VERTEX_SHADER, vertCode)
        val fs = try { compile(GLES20.GL_FRAGMENT_SHADER, fragCode) }
            catch (failure: Exception) { GLES20.glDeleteShader(vs); throw failure }
        val program = GLES20.glCreateProgram()
        try {
            GLES20.glAttachShader(program, vs); GLES20.glAttachShader(program, fs)
            GLES20.glLinkProgram(program)
        } finally { GLES20.glDeleteShader(vs); GLES20.glDeleteShader(fs) }
        val status = IntArray(1)
        GLES20.glGetProgramiv(program, GLES20.GL_LINK_STATUS, status, 0)
        check(status[0] != 0) { "Visual shader link failed: ${GLES20.glGetProgramInfoLog(program).take(512)}" }
        return program
    }

    private fun compile(type: Int, code: String): Int {
        val shader = GLES20.glCreateShader(type)
        GLES20.glShaderSource(shader, code); GLES20.glCompileShader(shader)
        val status = IntArray(1)
        GLES20.glGetShaderiv(shader, GLES20.GL_COMPILE_STATUS, status, 0)
        if (status[0] == 0) {
            val message = GLES20.glGetShaderInfoLog(shader).take(512)
            GLES20.glDeleteShader(shader)
            error("Visual shader compile failed: $message")
        }
        return shader
    }

    private companion object {
        const val MAX_POINTS = 15360
        const val BACKGROUND_POINTS = 256
    }
}

// ─────────────────────────────────────────────────────────────────────────────
// RIKO DNA (MDNA) — SYNC HUB (P1)
//
// Owns the audio-thread → render-thread pipeline:
//
//   PcmAudioProcessor (audio thread, ~60Hz, UNdelayed)
//        │  RawSpectrumTap.onFrame
//        ▼
//   SpectrumHistoryRing  ──►  LatencyCompensatedSource (render thread, every vsync)
//                                   asks for the spectrum at  now − latency
//
// LATENCY MODEL (honest version): the tap runs ahead of the speaker by the
// output latency. Media3 exposes NO continuous output-latency reading, so
// latency = a base estimate (the project's existing AV_SYNC_DELAY_MS, so MDNA
// lines up with the other visualizers) + a USER OFFSET that is remembered
// separately for built-in/wired vs Bluetooth output (Bluetooth lag is large
// and differs per headset — only the listener can calibrate it).
//
// The tap is attached ONLY while an MDNA surface is on screen (ref-counted), so
// every other screen pays nothing.
// ─────────────────────────────────────────────────────────────────────────────
@Singleton
class MdnaSyncHub @Inject constructor(
    private val pcm: PcmAudioProcessor,
    private val player: ExoPlayer,
    private val experienceDataHub: ExperienceDataHub,
    @ApplicationContext private val context: Context
) : RawSpectrumTap, LatencyProvider {

    val ring = SpectrumHistoryRing(SpectrumBands.COUNT)

    /** The production [BandSource]: history ring + latency compensation. */
    private val compensated = LatencyCompensatedSource(ring, this)
    data class TimelineInput(val songId: Long, val timeline: com.rikodev.music.android.dna.analysis.DnaTimeline)
    private data class Anchor(val position: Long, val speed: Float, val nanos: Long, val songId: Long?)
    @Volatile var timelineInput: TimelineInput? = null
    @Volatile private var anchor = Anchor(0L, 1f, 0L, null)
    private var blend = 0f // Render-thread only.
    private var renderNanos = 0L
    @Volatile private var outputGain = 1f
    @Volatile var liveSpectrumAvailable = false
        private set
    @Volatile var offlineTimelineAvailable = false
        private set
    /** Live dots never start a decoder or substitute a cached three-band DNA timeline. */
    val liveOnlySource: BandSource = object : BandSource {
        override val bandCount = SpectrumBands.COUNT
        override fun sample(nowNanos: Long, out: FloatArray): Boolean {
            val available = playing && compensated.sample(nowNanos, out)
            if (!available) out.fill(0f) else applyOutputGain(out)
            liveSpectrumAvailable = available
            offlineTimelineAvailable = false
            return available
        }
    }
    val source: BandSource = object : BandSource {
        override val bandCount = SpectrumBands.COUNT
        override fun sample(nowNanos: Long, out: FloatArray): Boolean {
            if (!playing) {
                out.fill(0f); blend = 0f; renderNanos = nowNanos
                liveSpectrumAvailable = false; offlineTimelineAvailable = false
                return false
            }
            val live = compensated.sample(nowNanos, out)
            val input = timelineInput
            val clock = anchor
            val offline = input?.takeIf { it.songId == clock.songId }?.timeline
            liveSpectrumAvailable = live
            offlineTimelineAvailable = offline != null
            val dt = if (renderNanos == 0L) 0f else ((nowNanos-renderNanos)/1e9f).coerceIn(0f,.1f)
            renderNanos = nowNanos
            val target = if (offline != null) (if (live) 0f else 1f) else 0f
            blend += (target-blend) * (1f-kotlin.math.exp(-dt*4f))
            if (offline == null || offline.frameCount == 0) { applyOutputGain(out); return live }
            // Media3 position is the presentation estimate; do not subtract decoder latency twice.
            val mediaUs = com.rikodev.music.android.dna.analysis.DnaTimelineSampler.mediaTimeUs(
                clock.position, clock.speed, clock.nanos, nowNanos, offsetMs)
            val bassEnd = SpectrumBands.bandIndex(250f, sampleRate.takeIf { it>0 } ?: 44100)
            val midEnd = SpectrumBands.bandIndex(4000f, sampleRate.takeIf { it>0 } ?: 44100)
            val bass = com.rikodev.music.android.dna.analysis.DnaTimelineSampler.sample(offline, mediaUs, 0)
            val mid = com.rikodev.music.android.dna.analysis.DnaTimelineSampler.sample(offline, mediaUs, 1)
            val treble = com.rikodev.music.android.dna.analysis.DnaTimelineSampler.sample(offline, mediaUs, 2)
            for (i in out.indices) {
                val value = if (i<bassEnd) bass else if (i<midEnd) mid else treble
                val count = if (i<bassEnd) bassEnd else if (i<midEnd) midEnd-bassEnd else out.size-midEnd
                val absolute = kotlin.math.sqrt(value.coerceAtLeast(0f) / kotlin.math.sqrt(count.coerceAtLeast(1).toFloat()))
                out[i] = out[i]*(1-blend)+absolute*blend
            }
            applyOutputGain(out)
            return true
        }
    }
    private fun applyOutputGain(out: FloatArray) {
        val gain = outputGain
        for (i in out.indices) out[i] *= gain
    }
    private fun updateAnchor() {
        // User volume estimate, not measured speaker SPL or a post-hardware-EQ capture.
        val maximum = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC).coerceAtLeast(1)
        val fraction = if (audioManager.isStreamMute(AudioManager.STREAM_MUSIC)) 0f
            else audioManager.getStreamVolume(AudioManager.STREAM_MUSIC).toFloat() / maximum
        outputGain = kotlin.math.sqrt((player.volume * fraction).coerceIn(0f,1f))
        anchor = Anchor(player.currentPosition, player.playbackParameters.speed, System.nanoTime(), player.currentMediaItem?.mediaId?.toLongOrNull())
    }
    @Volatile var playing = false
        private set
    @Volatile var positionMs = 0L
        private set
    @Volatile var playbackSpeed = 1f
        private set
    @Volatile var lastFrameNanos = 0L
        private set
    @Volatile var framesReceived = 0L
        private set
    val sampleRate: Int get() = pcm.diagnostics.value.sampleRate
    private val handler = Handler(Looper.getMainLooper())
    private val audioManager get() = context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
    private val clockUpdate = object : Runnable {
        override fun run() {
            updateAnchor()
            positionMs = player.currentPosition
            playbackSpeed = player.playbackParameters.speed
            if (attached.get() > 0) handler.postDelayed(this, 100L)
        }
    }
    private val playerListener = object : Player.Listener {
        override fun onEvents(player: Player, events: Player.Events) {
            if (events.contains(Player.EVENT_POSITION_DISCONTINUITY) ||
                events.contains(Player.EVENT_MEDIA_ITEM_TRANSITION) ||
                events.contains(Player.EVENT_PLAYBACK_PARAMETERS_CHANGED) ||
                events.contains(Player.EVENT_IS_PLAYING_CHANGED)) {
                onReset()
            }
            playing = player.isPlaying
            updateAnchor()
            positionMs = player.currentPosition
            playbackSpeed = player.playbackParameters.speed
        }
    }
    private val routeListener = object : AudioDeviceCallback() {
        override fun onAudioDevicesAdded(addedDevices: Array<out AudioDeviceInfo>) = routeChanged()
        override fun onAudioDevicesRemoved(removedDevices: Array<out AudioDeviceInfo>) = routeChanged()
    }
    private fun routeChanged() {
        bluetooth = isBluetoothOutput()
        onReset()
    }

    private val prefs: SharedPreferences by lazy {
        context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
    }
    private val attached = AtomicInteger(0)

    @Volatile private var loaded = false
    @Volatile private var bluetooth = false
    @Volatile private var offsetBuiltinMs = 0
    @Volatile private var offsetBluetoothMs = 0

    /** Base estimate shared with the other visualizers (ms). */
    val baseLatencyMs: Int get() = SpectrumConfig.AV_SYNC_DELAY_MS.toInt()

    /** The user's correction for the CURRENT output kind (ms, may be negative). */
    val offsetMs: Int
        get() {
            ensureLoaded()
            return if (bluetooth) offsetBluetoothMs else offsetBuiltinMs
        }

    val outputLabel: String get() = if (bluetooth) "Bluetooth connected (estimate)" else "Speaker / wired (estimate)"

    override fun latencyNanos(): Long = (baseLatencyMs + offsetMs).coerceAtLeast(0) * 1_000_000L

    fun setOffsetMs(ms: Int) {
        ensureLoaded()
        val v = ms.coerceIn(MIN_OFFSET_MS, MAX_OFFSET_MS)
        if (bluetooth) {
            offsetBluetoothMs = v
            prefs.edit().putInt(KEY_BLUETOOTH, v).apply()
        } else {
            offsetBuiltinMs = v
            prefs.edit().putInt(KEY_BUILTIN, v).apply()
        }
    }

    /** Ref-counted: every on-screen MDNA surface calls attach() / detach() once. */
    fun attach() {
        ensureLoaded()
        bluetooth = isBluetoothOutput()
        check(Looper.myLooper() == Looper.getMainLooper())
        if (attached.getAndIncrement() == 0) {
            onReset()
            playing = player.isPlaying
            player.addListener(playerListener)
            audioManager.registerAudioDeviceCallback(routeListener, handler)
            handler.post(clockUpdate)
            pcm.rawTap = this
        }
    }

    fun detach() {
        val remaining = attached.updateAndGet { if (it > 0) it - 1 else 0 }
        if (remaining == 0) {
            if (pcm.rawTap === this) pcm.rawTap = null
            playing = false
            player.removeListener(playerListener)
            audioManager.unregisterAudioDeviceCallback(routeListener)
            handler.removeCallbacks(clockUpdate)
            onReset()
        }
    }

    // ── RawSpectrumTap (AUDIO THREAD — allocation-free, never throws) ──

    override fun onFrame(levels: FloatArray, timestampNanos: Long) {
        ring.publish(levels, timestampNanos)
        // Phase 2: live sound-type classification feeds AudioState.soundClass.
        experienceDataHub.feedSpectrum(levels, sampleRate, timestampNanos)
        lastFrameNanos = timestampNanos
        framesReceived++
    }

    override fun onReset() {
        ring.markReset(System.nanoTime())
    }

    // ── internals ──

    private fun ensureLoaded() {
        if (loaded) return
        offsetBuiltinMs = prefs.getInt(KEY_BUILTIN, 0)
        offsetBluetoothMs = prefs.getInt(KEY_BLUETOOTH, 0)
        loaded = true
    }

    private fun isBluetoothOutput(): Boolean {
        val am = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager ?: return false
        return am.getDevices(AudioManager.GET_DEVICES_OUTPUTS).any { d ->
            d.type == AudioDeviceInfo.TYPE_BLUETOOTH_A2DP ||
                (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S &&
                    (d.type == AudioDeviceInfo.TYPE_BLE_HEADSET || d.type == AudioDeviceInfo.TYPE_BLE_SPEAKER))
        }
    }

    companion object {
        const val MIN_OFFSET_MS = -150
        const val MAX_OFFSET_MS = 400
        private const val PREFS_NAME = "mdna_sync"
        private const val KEY_BUILTIN = "offset_builtin_ms"
        private const val KEY_BLUETOOTH = "offset_bluetooth_ms"
    }
}

@Composable
fun DnaSharePanel(genome: DnaGenome?, song: Song?, viewModel: DnaLabViewModel, onImport: (DnaGenome) -> Unit) {
    var open by remember { mutableStateOf(false) }
    var code by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }
    var percent by remember { mutableIntStateOf(0) }
    var task by remember { mutableStateOf<Job?>(null) }
    val scope = rememberCoroutineScope()
    val context = LocalContext.current
    fun action(body: () -> Unit) { try { body() } catch (_: Exception) { message = "Invalid or unsupported DNA code" } }
    fun export(video: Boolean) {
        val dna = genome ?: return
        task = scope.launch {
            try {
                message = "Exporting…"; percent = 0
                val timeline = if (video) song?.let { viewModel.timeline(it) } else null
                val file = DnaExporter.export(context, dna, timeline, video) { value -> percent = value }
                val uri = FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
                val intent = Intent(Intent.ACTION_SEND).apply {
                    type = if(video) "video/mp4" else "image/png"
                    putExtra(Intent.EXTRA_STREAM, uri); clipData = ClipData.newRawUri("RIKO DNA", uri)
                    addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                }
                context.startActivity(Intent.createChooser(intent, "Share Music DNA"))
                message = "Export ready"
            } catch (cancel: CancellationException) { message = "Export cancelled"; throw cancel }
            catch (_: Exception) { message = "Export unavailable. Try PNG, or finish song analysis before video export." }
            finally { task = null }
        }
    }
    TextButton(onClick = { open = true }) { Text("Share / Import DNA") }
    if (open) AlertDialog(onDismissRequest = { task?.cancel(); open = false },
        title = { Text("Music DNA") },
        text = {
            Column(Modifier.verticalScroll(rememberScrollState())) {
                Text("MP4 uses the saved three-band timeline. It is not a recording of the live 40-band spectrum.", style = MaterialTheme.typography.bodySmall)
                OutlinedTextField(value = code, onValueChange = { code = it.take(180) }, label = { Text("RIKO1 DNA code") })
                Row {
                    TextButton(enabled = genome != null && task == null, onClick = { action {
                        (context.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager)
                            .setPrimaryClip(ClipData.newPlainText("Music DNA", DnaCode.encode(genome!!)))
                        message = "DNA code copied"
                    } }) { Text("Copy") }
                    TextButton(enabled = task == null, onClick = { action { onImport(DnaCode.decode(code)); message = "Imported DNA identity" } }) { Text("Import") }
                }
                Row {
                    TextButton(enabled = genome != null && task == null, onClick = { action {
                        message = "Visual trait distance: ${(DnaCode.distance(genome!!, DnaCode.decode(code))*100).toInt()}%"
                    } }) { Text("Compare") }
                    TextButton(enabled = genome != null && task == null, onClick = { action {
                        onImport(DnaCode.breed(genome!!, DnaCode.decode(code))); message = "Offspring DNA identity ready"
                    } }) { Text("Breed") }
                }
                Row {
                    TextButton(enabled = genome != null && task == null, onClick = { export(false) }) { Text("PNG card") }
                    TextButton(enabled = genome != null && song != null && task == null, onClick = { export(true) }) { Text("10s MP4") }
                }
                Text(if (task != null) "$message $percent%" else message)
                if (task != null) TextButton(onClick = { task?.cancel() }) { Text("Cancel export") }
            }
        }, confirmButton = { TextButton(onClick = { task?.cancel(); open = false }) { Text("Close") } })
}

import androidx.lifecycle.LifecycleEventObserver
import com.rikodev.music.android.dna.render.visual.RikoVisual
import com.rikodev.music.android.dna.render.visual.BaseVisualRenderer
import com.rikodev.music.android.dna.render.MdnaQuality
import com.rikodev.music.android.dna.sync.MdnaSyncHub

/**
 * Generic host for any RIKO visual. Creates the renderer, attaches the PCM
 * tap so real audio data flows, and manages GL lifecycle robustly.
 */
@Composable
fun VisualHost(
    visual: RikoVisual,
    hub: MdnaSyncHub,
    playing: Boolean,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val owner = LocalLifecycleOwner.current
    val view = remember(context, visual.id) {
        @Suppress("DEPRECATION")
        val refresh = (context.getSystemService(Context.WINDOW_SERVICE) as? WindowManager)
            ?.defaultDisplay?.refreshRate ?: 60f
        val renderer = visual.factory(hub.liveOnlySource, refresh) as BaseVisualRenderer
        VisualGLSurfaceView(context, renderer).also {
            if ((context.getSystemService(Context.ACTIVITY_SERVICE) as? ActivityManager)?.isLowRamDevice == true) {
                renderer.quality = MdnaQuality.LOW
            }
        }
    }

    // Attach PCM tap + ensure GL thread is running.
    DisposableEffect(owner, view) {
        hub.attach()
        val observer = LifecycleEventObserver { _, event ->
            when (event) {
                Lifecycle.Event.ON_RESUME -> { view.onResume() }
                Lifecycle.Event.ON_PAUSE -> { view.onPause() }
                else -> {}
            }
        }
        owner.lifecycle.addObserver(observer)
        // Always resume — GLSurfaceView needs onResume() to start the GL thread.
        view.onResume()
        onDispose {
            owner.lifecycle.removeObserver(observer)
            view.onPause()
            hub.detach()
        }
    }

    // Keep renderer state in sync with playback + surface error.
    var glError by remember { mutableStateOf<String?>(null) }
    LaunchedEffect(playing) {
        (view.renderer as? BaseVisualRenderer)?.playing = playing
        while (true) {
            glError = (view.renderer as? BaseVisualRenderer)?.error
            kotlinx.coroutines.delay(1000)
        }
    }

    Box(modifier = modifier) {
        AndroidView(factory = { view }, modifier = Modifier.fillMaxSize())
        glError?.let { msg ->
            Text(
                "GL Error: $msg",
                color = Color(0xFFFF6666),
                fontSize = 11.sp,
                modifier = Modifier.align(Alignment.Center).padding(16.dp)
            )
        }
    }
}
